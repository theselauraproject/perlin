cmake_minimum_required(VERSION 3.24)
project(perlin_api LANGUAGES C CXX)

include("${CMAKE_CURRENT_SOURCE_DIR}/cmake/PerlinApi.cmake")

option(PERLIN_API_BUILD_TESTS "Build the SDK and wire tests" ON)
option(PERLIN_API_BUILD_HOST "Build the shared host core on wasm3" OFF)

if(PERLIN_API_BUILD_HOST)
    include(FetchContent)
    FetchContent_Declare(wasm3 GIT_REPOSITORY https://github.com/wasm3/wasm3.git GIT_TAG 28ecb9af6d2040e474a70f7cb7f43666740141fb)
    FetchContent_GetProperties(wasm3)
    if(NOT wasm3_POPULATED)
        FetchContent_Populate(wasm3)
    endif()
    file(GLOB PERLIN_API_WASM3_SOURCES "${wasm3_SOURCE_DIR}/source/*.c")
    list(FILTER PERLIN_API_WASM3_SOURCES EXCLUDE REGEX "m3_api_")
    add_library(perlin_api_wasm3 STATIC ${PERLIN_API_WASM3_SOURCES})
    target_include_directories(perlin_api_wasm3 PUBLIC "${wasm3_SOURCE_DIR}/source")
    target_compile_definitions(perlin_api_wasm3 PRIVATE d_m3HasTracer=0)
    target_compile_options(perlin_api_wasm3 PRIVATE -w)
    add_library(perlin_api_host STATIC ${PERLIN_API_HOST_CORE_SOURCES} ${PERLIN_API_HOST_WASM3_SOURCES})
    target_link_libraries(perlin_api_host PUBLIC perlin_api perlin_api_wasm3)
endif()

if(PERLIN_API_BUILD_TESTS AND NOT MSVC)
    enable_testing()
    perlin_api_add_checks()
    set(PERLIN_API_GEN "${CMAKE_CURRENT_BINARY_DIR}/gen")
    add_custom_command(OUTPUT "${PERLIN_API_GEN}/sdk/mod_settings.hpp"
        COMMAND ${CMAKE_COMMAND} -E make_directory "${PERLIN_API_GEN}/sdk"
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_CURRENT_SOURCE_DIR}/tools/perlin_settings.py"
                "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/settings_test.json" "${PERLIN_API_GEN}/sdk/mod_settings.hpp"
        DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/tests/data/settings_test.json" "${CMAKE_CURRENT_SOURCE_DIR}/tools/perlin_settings.py")
    add_executable(perlin_api_sdk_test tests/sdk_test.cpp "${PERLIN_API_GEN}/sdk/mod_settings.hpp")
    target_include_directories(perlin_api_sdk_test PRIVATE "${PERLIN_API_GEN}/sdk")
    target_link_libraries(perlin_api_sdk_test PRIVATE perlin_api)
    add_test(NAME perlin_api_sdk_test COMMAND perlin_api_sdk_test)
    foreach(t abi_test meta_test plugin_test)
        add_executable(perlin_api_${t} tests/${t}.cpp)
        target_link_libraries(perlin_api_${t} PRIVATE perlin_api)
        add_test(NAME perlin_api_${t} COMMAND perlin_api_${t})
    endforeach()
    add_executable(perlin_api_c_header_test tests/c_header_test.c)
    target_include_directories(perlin_api_c_header_test PRIVATE ${PERLIN_API_INCLUDE_DIRS})
    set_target_properties(perlin_api_c_header_test PROPERTIES C_STANDARD 99)
    add_test(NAME perlin_api_c_header_test COMMAND perlin_api_c_header_test)
endif()
